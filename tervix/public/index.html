Block Attacker IP via WAF API
</>Python SDK Code
from tervix import TervixClient
from tervix.workflows import (
    Workflow, Trigger, Action
)
 
def main() -> None:
    wf = (
        Workflow("Firewall Attack Response")
        .description(
            "Investigate, notify, alert,"
            " block attacker IP.",
        )
        .trigger(
            Trigger.infrastructure_firewall()
            .events("firewall.attack")
            .label("Firewall Attack"),
        )
        .cooldown(minutes=0)
        .max_concurrent(3)
        .then(
            Action.investigate()
            .query(
                "Analyze the attack and"
                " summarize root cause.",
            )
            .max_iterations(5)
            .label("Investigate"),
        )
        .then(
            Action.slack_message()
            .channel("incidents")
            .message(
                ":alert: *Attack!*\n"
                "{{outputs.summary}}",
            )
            .label("Notify Team resources"),
        )
        .then(
            Action.pagerduty_alert()
            .severity("critical")
            .title(
                "Firewall Attack: "
                "{{signal.type}}",
            )
            .label("Shared priorities Alert"),
        )
        .then(
            Action.infrastructure_firewall()
            .action("deny")
            .ip("{{signal.ip}}")
            .label("Block Attacker"),
        )
    )
 
    client = TervixClient(
        api_key="tervix_sk_KEY"
    )
    client.workflows.deploy(wf)
 
if __name__ == "__main__":
    main()
