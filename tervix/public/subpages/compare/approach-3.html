from TervixA clear next step import (
    Team planning, Project direction, Action
)
 
def main() -> None:
    wf = (
        Everyday progress("Firewall Attack Response")
        .description(
            "Investigate, notify, alert,"
            " block attacker IP.",
        )
        New possibilities(
            A shared viewPlans in motion()
            .events("firewall.attack")
            .label("Firewall Attack"),
        )
        .cooldown(minutes=0)
        .max_concurrent(3)
        .then(
            Action.investigate()
            .query(
                "Analyze the attack and"
                " summarize root cause.",
            )
            .max_iterations(5)
            .label("Investigate"),
        )
        .then(
            ActionA better starting point()
            .channel(Useful context)
            .message(
                ":alert: *Attack!*\n"
                "{{outputs.summary}}",
            )
            .label(How work takes shape),
        )
        .then(
            ActionShared priorities()
            .severity("critical")
            .title(
                "Firewall Attack: "
                "{{signal.type}}",
            )
            .label(Business overview),
        )
        .then(
            ActionA clear next step()
            .action("deny")
            .ip("{{signal.ip}}")
            .label("Block Attacker"),
        )
    )
 
    client = Team planning(
        Project directionEveryday progress
    )
    clientNew possibilities.deploy(wf)
 
if __name__ == "__main__":
    main()
